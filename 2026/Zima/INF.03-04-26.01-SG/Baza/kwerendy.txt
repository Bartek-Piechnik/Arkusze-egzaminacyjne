1. SELECT id, imie, nazwisko FROM maturzysta WHERE szkola = "T3" ORDER BY nazwisko ASC;
2. SELECT DISTINCT przedmiot FROM arkusz;
3. SELECT MIN(rok) AS "Najstarszy arkusz", MAX(rok) AS "Najmłodszy arkusz" FROM arkusz;
4. SELECT m.id, AVG(punkty) AS "Wynik" FROM maturzysta m JOIN wynik w ON m.id = w.maturzysta_id GROUP BY m.id ORDER BY AVG(punkty) DESC LIMIT 1;
5. SELECT a.rok, a.sesja, a.przedmiot, w.punkty FROM arkusz a JOIN wynik w ON a.symbol = w.symbol WHERE w.maturzysta_id = 31;
