2. SELECT nazwa FROM produkty WHERE cena_brutto > 500;
3. SELECT k.nazwa_kategorii, AVG(p.cena_brutto) FROM produkty p JOIN kategorie k on p.id_kat = k.id_kat GROUP BY k.nazwa_kategorii;
